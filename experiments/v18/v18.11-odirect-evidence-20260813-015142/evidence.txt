 13797: static void cpu_v19_aio_poll(void) {
 13798:     if (!g_v19_aio.ready) {
 13799:         return;
 13800:     }
 13801: 
 13802:     for (;;) {
 13803: 
 13804:         struct io_event ev[16];
 13805: 
 13806:         struct timespec ts = {
 13807:             .tv_sec = 0,
 13808:             .tv_nsec = 0,
 13809:         };
 13810: 
 13811:         long n =
 13812:             cpu_v19_io_getevents(
 13813:                 g_v19_aio.ctx,
 13814:                 0,
 13815:                 16,
 13816:                 ev,
 13817:                 &ts);
 13818: 
 13819:         if (n <= 0) {
 13820:             break;
 13821:         }
 13822: 
 13823:         for (long i = 0;
 13824:              i < n;
 13825:              i++) {
 13826: 
 13827:             v19_aio_op *op =
 13828:                 (v19_aio_op *)
 13829:                 (uintptr_t)ev[i].data;
 13830: 
 13831:             if (!op) {
 13832:                 g_v19_aio.io_errors++;
 13833:                 continue;
 13834:             }
 13835: 
 13836:             v19_aio_slot *slot =
 13837:                 cpu_v19_slot_from_op(op);
 13838: 
 13839:             if (!slot) {
 13840:                 g_v19_aio.io_errors++;
 13841:                 continue;
 13842:             }
 13843: 
 13844:             if (op->submitted &&
 13845:                 !op->done) {
 13846: 
 13847:                 op->done = true;
 13848: 
 13849:                 op->ok =
 13850:                     ev[i].res ==
 13851:                     (int64_t)op->bytes;
 13852: 
 13853:                 if (op->ok) {
 13854:                     g_v19_aio.completed_ops++;
 13855: 
 13856:                     g_v19_aio.completed_bytes +=
 13857:                         op->bytes;
 13858:                 } else {
 13859:                     g_v19_aio.io_errors++;
 13860:                 }
 13861: 
 13862:                 if (slot->pending > 0) {
 13863:                     slot->pending--;
 13864:                 }
 13865:             }
 13866: 
 13867:             if (slot->released &&
 13868:                 slot->pending == 0) {
 13869: 
 13870:                 slot->active =
 13871:                     false;
 13872:             }
 13873:         }
 13874:     }
 13875: }
